# PARP14 Domain Deletion Library - CALVADOS Simulations

## Overview

High-throughput coarse-grained MD simulation study of PARP14 domain deletion variants to understand how domain composition affects structural dynamics.

**This is Phase 2** of the PARP14 computational pipeline. Phase 1 (structure generation with AlphaFold3) is documented in [`../parp14/README.md`](../parp14/README.md).

│                                  │     │                                  │
│  • AlphaFold3 predictions        │     │  • CALVADOS coarse-grained MD    │

| Production (keep) | 40 ns |
| Compiled per composition | 1 μs (25 × 40 ns) |

## Simulation Protocol

### Per-Structure Workflow

1. Load PDB structure from Phase 1 outputs
2. Energy minimization (automatic in CALVADOS)
3. 50 ns production MD (5,000,000 steps)
4. Save trajectory every 100 ps (wfreq=10000)
5. Output: 500 frames per simulation

### Per-Composition Workflow

1. Run 25 replicate simulations (different random seeds)
2. Discard first 100 frames (10 ns equilibration) from each
3. Keep last 400 frames (40 ns production) from each
4. Concatenate: 25 × 400 = 10,000 frames = 1 μs compiled trajectory

### Batch Preparation Script

```python
#!/usr/bin/env python
"""prepare_variant.py - Prepare simulation for one composition + replicate"""

import os
import sys
from calvados.cfg import Config, Components

def prepare_variant(comp_name, rep_id, base_path):
    """Prepare simulation files for one variant structure."""

    rep_str = f"rep_{rep_id:02d}"
    sysname = f"parp14_{comp_name}_s{rep_id:02d}"

    sim_path = f"{base_path}/simulations/{comp_name}/{rep_str}"
    os.makedirs(sim_path, exist_ok=True)

    config = Config(
        sysname = sysname,
        box = [100, 100, 100],
        temp = 293,
        ionic = 0.19,
        pH = 7.0,
        steps = 5000000,
        wfreq = 10000,
        platform = 'CUDA',
        restart = None,
        verbose = False,
    )

    components = Components(
        fresidues = f'{base_path}/input/residues_CALVADOS3.csv',
        fdomains = f'{base_path}/input/domains/domains_{comp_name}.yaml',
        pdb_folder = f'{base_path}/input/structures/{comp_name}',
        restraint_type = 'harmonic',
        k_harmonic = 700.,
        use_com = True,
    )

    # Add the structure (use first PDB in folder)
    pdb_files = [f for f in os.listdir(f'{base_path}/input/structures/{comp_name}')
                 if f.endswith('.pdb')]
    struct_name = pdb_files[0].replace('.pdb', '')
    components.add(name=struct_name, restraint=True)

    config.write(sim_path)
    components.write(sim_path)

    print(f"Prepared: {sim_path}")
    return sim_path

if __name__ == "__main__":
    comp_name = sys.argv[1]           # e.g., 'rrm1_rrm2_rrm3'
    rep_id = int(sys.argv[2])         # e.g., 1-25
    base_path = sys.argv[3] if len(sys.argv) > 3 else os.getcwd()
    prepare_variant(comp_name, rep_id, base_path)
```

source ~/.bashrc
conda activate calvados

cd $SIM_PATH
python run.py --path .

echo "Completed: $SIM_PATH"

### Trajectory Compilation

```python
#!/usr/bin/env python
"""compile_trajectories.py - Merge 25 replicates into 1 μs trajectory"""

import MDAnalysis as mda
from MDAnalysis import Writer
import numpy as np
import os
import sys

def compile_composition(comp_name, base_path, n_reps=25, skip_frames=100):
    """
    Compile 25 replicates, skipping equilibration.

    skip_frames=100 corresponds to 10 ns when wfreq=10000 (100 ps/frame)
    """

    sim_dir = f"{base_path}/simulations/{comp_name}"
    out_dir = f"{base_path}/trajectories"
    os.makedirs(out_dir, exist_ok=True)

    # Use topology from first replicate
    top_pdb = f"{sim_dir}/rep_01/top.pdb"

    # Output files
    out_dcd = f"{out_dir}/{comp_name}.dcd"
    out_top = f"{out_dir}/{comp_name}_top.pdb"

    # Copy topology
    u_top = mda.Universe(top_pdb)
    u_top.atoms.write(out_top)

    total_frames = 0

    with Writer(out_dcd, n_atoms=u_top.atoms.n_atoms) as W:
        for rep in range(1, n_reps + 1):
            rep_dir = f"{sim_dir}/rep_{rep:02d}"

            # Find trajectory file
            traj_files = [f for f in os.listdir(rep_dir) if f.endswith('.dcd')]
            if not traj_files:
                print(f"Warning: No trajectory found in {rep_dir}")
                continue

            traj_path = f"{rep_dir}/{traj_files[0]}"
            u = mda.Universe(top_pdb, traj_path)

            # Skip equilibration, write production frames
            for ts in u.trajectory[skip_frames:]:
                W.write(u.atoms)
                total_frames += 1

    print(f"Compiled {comp_name}: {total_frames} frames from {n_reps} replicates")
    print(f"  Output: {out_dcd}")

    return out_dcd, total_frames

if __name__ == "__main__":
    comp_name = sys.argv[1]
    base_path = sys.argv[2] if len(sys.argv) > 2 else os.getcwd()
    compile_composition(comp_name, base_path)
```

### Analysis Script

```python
#!/usr/bin/env python
"""analyze_composition.py - Analyze compiled trajectory for one composition"""

import MDAnalysis as mda
import numpy as np
import pandas as pd
from calvados.analysis import calc_rg, calc_ete, calc_cmap, fit_scaling_exp
from calvados.BLOCKING.main import BlockAnalysis
import sys
import os

def analyze_composition(comp_name, base_path):
    """Analyze compiled 1 μs trajectory."""

    traj_dir = f"{base_path}/trajectories"
    out_dir = f"{base_path}/analysis/per_composition"
    os.makedirs(out_dir, exist_ok=True)
    os.makedirs(f"{base_path}/analysis/contact_maps", exist_ok=True)
    os.makedirs(f"{base_path}/analysis/rg_distributions", exist_ok=True)

    # Load trajectory
    top = f"{traj_dir}/{comp_name}_top.pdb"
    traj = f"{traj_dir}/{comp_name}.dcd"

    u = mda.Universe(top, traj)
    ag = u.select_atoms('all')

    # Load residue parameters
    residues = pd.read_csv(f'{base_path}/input/residues_CALVADOS3.csv').set_index('three')

    results = {'composition': comp_name}

    # Radius of gyration
    rgs = calc_rg(u, ag, ag.resnames.tolist(), residues)
    block_rg = BlockAnalysis(rgs)
    block_rg.SEM()
    results['rg_mean'] = np.mean(rgs)
    results['rg_std'] = np.std(rgs)
    results['rg_sem'] = block_rg.sem

    # End-to-end distance
    rees, ree_mean, ree_sem = calc_ete(u, ag)
    results['ree_mean'] = ree_mean
    results['ree_sem'] = ree_sem

    # Contact map (average)
    cmap = np.zeros((len(ag), len(ag)))
    for ts in u.trajectory:
        cmap += calc_cmap(ag, ag)
    cmap /= len(u.trajectory)
    np.save(f"{base_path}/analysis/contact_maps/{comp_name}_cmap.npy", cmap)

    # Save Rg distribution
    np.save(f"{base_path}/analysis/rg_distributions/{comp_name}_rg.npy", rgs)

    # Save metrics
    df = pd.DataFrame([results])
    df.to_csv(f"{out_dir}/{comp_name}_metrics.csv", index=False)

    print(f"Analyzed {comp_name}: Rg={results['rg_mean']:.2f}±{results['rg_sem']:.2f} nm")

    return results

if __name__ == "__main__":
    comp_name = sys.argv[1]
    base_path = sys.argv[2] if len(sys.argv) > 2 else os.getcwd()
    analyze_composition(comp_name, base_path)
```

## Analysis Metrics

For each composition, compute:

| Metric | Description | Function |
|--------|-------------|----------|
| **Rg** | Radius of gyration | `calc_rg()` |
| **Ree** | End-to-end distance | `calc_ete()` |
| **Contact map** | Average inter-residue contacts | `calc_cmap()` |
| **RMSF** | Per-residue fluctuations | Custom |
| **Domain distances** | Pairwise domain COM distances | Custom |
| **ν (scaling)** | Polymer scaling exponent | `fit_scaling_exp()` |
| **Compactness** | Rg/Ree ratio | Computed |

For library-wide analysis:

- Rg vs domain composition
- Correlation with Phase 1 pLDDT scores
- Contact map clustering
- PCA of structural ensembles
- Domain deletion effects on flexibility
